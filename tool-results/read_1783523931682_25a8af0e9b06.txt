   147→// Deterministic hue per company name for avatar colour
   148→function companyHue(name: string): string {
   149→  let hash = 0;
   150→  for (let i = 0; i < name.length; i++) {
   151→    hash = name.charCodeAt(i) + ((hash << 5) - hash);
   152→  }
   153→  const hue = Math.abs(hash) % 360;
   154→  return `hsl(${hue}, 55%, 50%)`;
   155→}

   161→function CircularAtsScore({ score }: { score: number }) {
   162→  const radius = 18;
   163→  const circumference = 2 * Math.PI * radius;
   164→  const offset = circumference - (score / 100) * circumference;
   165→
   166→  const colour =
   167→    score >= 80
   168→      ? 'text-emerald-500'
   169→      : score >= 60
   170→        ? 'text-amber-500'
   171→        : 'text-red-500';
   172→
   173→  return (
   174→    <div className="relative inline-flex size-10 items-center justify-center">
   175→      <svg className="size-10 -rotate-90" viewBox="0 0 44 44">
   176→        <circle
   177→          cx="22"
   178→          cy="22"
   179→          r={radius}
   180→          fill="none"
   181→          stroke="currentColor"
   182→          strokeWidth="3"
   183→          className="text-muted/30"
   184→        />
   185→        <circle
   186→          cx="22"
   187→          cy="22"
   188→          r={radius}
   189→          fill="none"
   190→          stroke="currentColor"
   191→          strokeWidth="3"
   192→          strokeLinecap="round"
   193→          strokeDasharray={circumference}
   194→          strokeDashoffset={offset}
   195→          className={colour}
   196→        />
   197→      </svg>
   198→      <span className={`absolute text-[10px] font-bold ${colour}`}>
   199→        {score}
   200→      </span>
   201→    </div>
   202→  );
   203→}

   540→function StatsSummary({ applications }: { applications: Application[] }) {
   541→  const total = applications.length;
   542→  const responded = applications.filter(
   543→    (a) =>
   544→      a.status === 'reviewing' ||
   545→      a.status === 'interview' ||
   546→      a.status === 'assessment' ||
   547→      a.status === 'offer' ||
   548→      a.status === 'rejected'
   549→  ).length;
   550→  const interviewed = applications.filter(
   551→    (a) =>
   552→      a.status === 'interview' ||
   553→      a.status === 'assessment' ||
   554→      a.status === 'offer'
   555→  ).length;
   556→  const offered = applications.filter(
   557→    (a) => a.status === 'offer'
   558→  ).length;
   559→
   560→  const stats = [
   561→    {
   562→      label: 'Total Applications',
   563→      value: total,
   564→      icon: Briefcase,
   565→      color: 'text-foreground',
   566→    },
   567→    {
   568→      label: 'Response Rate',
   569→      value: total > 0 ? `${Math.round((responded / total) * 100)}%` : '0%',
   570→      icon: TrendingUp,
   571→      color: 'text-teal-600 dark:text-teal-400',
   572→    },
   573→    {
   574→      label: 'Interview Rate',
   575→      value:
   576→        total > 0 ? `${Math.round((interviewed / total) * 100)}%` : '0%',
   577→      icon: Video,
   578→      color: 'text-emerald-600 dark:text-emerald-400',
   579→    },
   580→    {
   581→      label: 'Offer Rate',
   582→      value: total > 0 ? `${Math.round((offered / total) * 100)}%` : '0%',
   583→      icon: Award,
   584→      color: 'text-green-600 dark:text-green-400',
   585→    },
   586→  ];
   587→
   588→  return (
   589→    <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   590→      {stats.map((s) => (
   591→        <Card key={s.label} className="py-0 gap-0">
   592→          <CardContent className="p-4 flex items-center gap-3">
   593→            <div
   594→              className={`rounded-lg bg-muted/60 p-2 ${s.color}`}
   595→            >
   596→              <s.icon className="size-4" />
   597→            </div>
   598→            <div>
   599→              <p className="text-lg font-bold leading-tight">{s.value}</p>
   600→              <p className="text-[11px] text-muted-foreground leading-tight">
   601→                {s.label}
   602→              </p>
   603→            </div>
   604→          </CardContent>
   605→        </Card>
   606→      ))}
   607→    </div>
   608→  );
   609→}

   629→    ghosted: 0,
   630→  });
   631→  const [loading, setLoading] = useState(true);
   632→
   633→  const fetchApplications = useCallback(async () => {
   634→    setLoading(true);
   635→    try {
   636→      const query =
   637→        selectedFilter === 'all' ? '' : `?status=${selectedFilter}`;
   638→      const res = await fetch(`/api/applications${query}`);
   639→      if (!res.ok) throw new Error('Failed to fetch applications');
   640→      const data = await res.json();
   641→
   642→      const apps: Application[] = data.applications ?? data ?? [];
   643→      const counts: StatusCounts = data.counts ?? {
   644→        all: 0,
   645→        submitted: 0,
   646→        reviewing: 0,
   647→        interview: 0,
   648→        assessment: 0,
   649→        offer: 0,
   650→        rejected: 0,
   651→        ghosted: 0,
   652→      };
   653→
   654→      // If counts not returned, compute them ourselves
   655→      if (!data.counts) {
   656→        const computed: Record<string, number> = {
   657→          all: apps.length,
   658→          submitted: 0,
   659→          reviewing: 0,
   660→          interview: 0,
   661→          assessment: 0,
   662→          offer: 0,
   663→          rejected: 0,
   664→          ghosted: 0,
   665→        };
   666→        for (const a of apps) {
   667→          if (computed[a.status] !== undefined) {
   668→            computed[a.status]++;
   669→          }
   670→        }
   671→        counts.all = computed.all;
   672→        counts.submitted = computed.submitted;
   673→        counts.reviewing = computed.reviewing;
   674→        counts.interview = computed.interview;
   675→        counts.assessment = computed.assessment;
   676→        counts.offer = computed.offer;
   677→        counts.rejected = computed.rejected;
   678→        counts.ghosted = computed.ghosted;
   679→      }
   680→
   681→      setApplications(apps);
   682→      setStatusCounts(counts);
   683→    } catch {
   684→      // On error, show empty state — silent fallback
   685→      setApplications([]);
   686→    } finally {
   687→      setLoading(false);
   688→    }
   689→  }, [selectedFilter]);
   690→
   691→  useEffect(() => {
   692→    fetchApplications();
   693→  }, [fetchApplications]);
   694→
   695→  const filteredApps = useMemo(() => {
   696→    if (selectedFilter === 'all') return applications;
   697→    return applications.filter((a) => a.status === selectedFilter);
   698→  }, [applications, selectedFilter]);
   699→
   700→  return (
   701→    <div className="flex flex-col gap-4 h-full">
   702→      {/* Header + View Toggle */}
   703→      <div className="flex items-center justify-between flex-wrap gap-3">
   704→        <h2 className="text-lg font-semibold flex items-center gap-2">
   705→          <Briefcase className="size-5" />
   706→          Applications
   707→        </h2>
   708→        <div className="flex items-center gap-1 bg-muted rounded-lg p-1">
   709→          <Button
   710→            variant={viewMode === 'card' ? 'default' : 'ghost'}
   711→            size="sm"
   712→            className="h-7 px-2.5 text-xs"
   713→            onClick={() => setViewMode('card')}
   714→          >
   715→            <LayoutGrid className="size-3.5 mr-1" />
   716→            Card View
   717→          </Button>
   718→          <Button
   719→            variant={viewMode === 'pipeline' ? 'default' : 'ghost'}
   720→            size="sm"
   721→            className="h-7 px-2.5 text-xs"
   722→            onClick={() => setViewMode('pipeline')}
   723→          >
   724→            <Columns className="size-3.5 mr-1" />
   725→            Pipeline View
   726→          </Button>
   727→        </div>
   728→      </div>
   729→
   730→      {/* Status Filter Bar */}
   731→      <ScrollArea className="w-full">
   732→        <div className="flex gap-2 pb-2 min-w-max">
   733→          {ALL_STATUSES.map((s) => {
   734→            const isActive = selectedFilter === s;
   735→            const label =
   736→              s === 'all'
   737→                ? 'All'
   738→                : STATUS_CONFIG[s as ApplicationStatus]?.label ?? s;
   739→            const count =
   740→              s === 'all'
   741→                ? statusCounts.all
   742→                : statusCounts[s as ApplicationStatus] ?? 0;
   743→
   744→            return (
   745→              <Button
   746→                key={s}
   747→                variant={isActive ? 'default' : 'outline'}
   748→                size="sm"
   749→                className="h-8 text-xs gap-1.5 rounded-full"
   750→                onClick={() => setSelectedFilter(s)}
   751→              >
   752→                {label}
   753→                <Badge
   754→                  variant="secondary"
   755→                  className={`ml-0.5 h-5 min-w-5 px-1.5 text-[10px] ${
   756→                    isActive
   757→                      ? 'bg-primary-foreground/20 text-primary-foreground'
   758→                      : ''
   759→                  }`}
   760→                >
   761→                  {count}
   762→                </Badge>
   763→              </Button>
   764→            );
   765→          })}
   766→        </div>
   767→        <ScrollBar orientation="horizontal" />
   768→      </ScrollArea>
   769→
   770→      {/* Content */}
   771→      <div className="flex-1 min-h-0">
   772→        {loading ? (
   773→          <div className="space-y-3">
   774→            {Array.from({ length: 4 }).map((_, i) => (
   775→              <SkeletonCard key={i} />
   776→            ))}
   777→          </div>
   778→        ) : filteredApps.length === 0 ? (
   779→          <div className="flex flex-col items-center justify-center py-16 text-center">
   780→            <div className="rounded-full bg-muted p-4 mb-4">
   781→              <Briefcase className="size-8 text-muted-foreground" />
   782→            </div>
   783→            <h3 className="font-medium text-sm">No applications found</h3>
   784→            <p className="text-xs text-muted-foreground mt-1">
   785→              {selectedFilter === 'all'
   786→                ? 'Start tracking your job applications.'
   787→                : `No applications with "${STATUS_CONFIG[selectedFilter as ApplicationStatus]?.label}" status.`}
   788→            </p>
   789→          </div>
   790→        ) : viewMode === 'card' ? (
   791→          <ScrollArea className="h-full">
   792→            <div className="space-y-3 pr-3">
   793→              {filteredApps.map((app) => (
   794→                <ApplicationCard key={app.id} app={app} />
   795→              ))}
   796→            </div>
   797→          </ScrollArea>
   798→        ) : (
   799→          <PipelineView applications={applications} />
   800→        )}
   801→      </div>
   802→
   803→      {/* Stats Summary */}
   804→      {!loading && applications.length > 0 && (
   805→        <Separator />
   806→      )}
   807→      {!loading && applications.length > 0 && (
   808→        <StatsSummary applications={applications} />
   809→      )}
   810→    </div>
   811→  );
   812→}
