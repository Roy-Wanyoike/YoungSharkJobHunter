   147→   147→// Deterministic hue per company name for avatar colour
   148→   148→function companyHue(name: string): string {
   149→   149→  let hash = 0;
   150→   150→  for (let i = 0; i < name.length; i++) {
   151→   151→    hash = name.charCodeAt(i) + ((hash << 5) - hash);
   152→   152→  }
   153→   153→  const hue = Math.abs(hash) % 360;
   154→   154→  return `hsl(${hue}, 55%, 50%)`;
   155→   155→}

   161→   161→function CircularAtsScore({ score }: { score: number }) {
   162→   162→  const radius = 18;
   163→   163→  const circumference = 2 * Math.PI * radius;
   164→   164→  const offset = circumference - (score / 100) * circumference;
   165→   165→
   166→   166→  const colour =
   167→   167→    score >= 80
   168→   168→      ? 'text-emerald-500'
   169→   169→      : score >= 60
   170→   170→        ? 'text-amber-500'
   171→   171→        : 'text-red-500';
   172→   172→
   173→   173→  return (
   174→   174→    <div className="relative inline-flex size-10 items-center justify-center">
   175→   175→      <svg className="size-10 -rotate-90" viewBox="0 0 44 44">
   176→   176→        <circle
   177→   177→          cx="22"
   178→   178→          cy="22"
   179→   179→          r={radius}
   180→   180→          fill="none"
   181→   181→          stroke="currentColor"
   182→   182→          strokeWidth="3"
   183→   183→          className="text-muted/30"
   184→   184→        />
   185→   185→        <circle
   186→   186→          cx="22"
   187→   187→          cy="22"
   188→   188→          r={radius}
   189→   189→          fill="none"
   190→   190→          stroke="currentColor"
   191→   191→          strokeWidth="3"
   192→   192→          strokeLinecap="round"
   193→   193→          strokeDasharray={circumference}
   194→   194→          strokeDashoffset={offset}
   195→   195→          className={colour}
   196→   196→        />
   197→   197→      </svg>
   198→   198→      <span className={`absolute text-[10px] font-bold ${colour}`}>
   199→   199→        {score}
   200→   200→      </span>
   201→   201→    </div>
   202→   202→  );
   203→   203→}

   540→   540→function StatsSummary({ applications }: { applications: Application[] }) {
   541→   541→  const total = applications.length;
   542→   542→  const responded = applications.filter(
   543→   543→    (a) =>
   544→   544→      a.status === 'reviewing' ||
   545→   545→      a.status === 'interview' ||
   546→   546→      a.status === 'assessment' ||
   547→   547→      a.status === 'offer' ||
   548→   548→      a.status === 'rejected'
   549→   549→  ).length;
   550→   550→  const interviewed = applications.filter(
   551→   551→    (a) =>
   552→   552→      a.status === 'interview' ||
   553→   553→      a.status === 'assessment' ||
   554→   554→      a.status === 'offer'
   555→   555→  ).length;
   556→   556→  const offered = applications.filter(
   557→   557→    (a) => a.status === 'offer'
   558→   558→  ).length;
   559→   559→
   560→   560→  const stats = [
   561→   561→    {
   562→   562→      label: 'Total Applications',
   563→   563→      value: total,
   564→   564→      icon: Briefcase,
   565→   565→      color: 'text-foreground',
   566→   566→    },
   567→   567→    {
   568→   568→      label: 'Response Rate',
   569→   569→      value: total > 0 ? `${Math.round((responded / total) * 100)}%` : '0%',
   570→   570→      icon: TrendingUp,
   571→   571→      color: 'text-teal-600 dark:text-teal-400',
   572→   572→    },
   573→   573→    {
   574→   574→      label: 'Interview Rate',
   575→   575→      value:
   576→   576→        total > 0 ? `${Math.round((interviewed / total) * 100)}%` : '0%',
   577→   577→      icon: Video,
   578→   578→      color: 'text-emerald-600 dark:text-emerald-400',
   579→   579→    },
   580→   580→    {
   581→   581→      label: 'Offer Rate',
   582→   582→      value: total > 0 ? `${Math.round((offered / total) * 100)}%` : '0%',
   583→   583→      icon: Award,
   584→   584→      color: 'text-green-600 dark:text-green-400',
   585→   585→    },
   586→   586→  ];
   587→   587→
   588→   588→  return (
   589→   589→    <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   590→   590→      {stats.map((s) => (
   591→   591→        <Card key={s.label} className="py-0 gap-0">
   592→   592→          <CardContent className="p-4 flex items-center gap-3">
   593→   593→            <div
   594→   594→              className={`rounded-lg bg-muted/60 p-2 ${s.color}`}
   595→   595→            >
   596→   596→              <s.icon className="size-4" />
   597→   597→            </div>
   598→   598→            <div>
   599→   599→              <p className="text-lg font-bold leading-tight">{s.value}</p>
   600→   600→              <p className="text-[11px] text-muted-foreground leading-tight">
   601→   601→                {s.label}
   602→   602→              </p>
   603→   603→            </div>
   604→   604→          </CardContent>
   605→   605→        </Card>
   606→   606→      ))}
   607→   607→    </div>
   608→   608→  );
   609→   609→}

   638→   638→      const res = await fetch(`/api/applications${query}`);
   639→   639→      if (!res.ok) throw new Error('Failed to fetch applications');
