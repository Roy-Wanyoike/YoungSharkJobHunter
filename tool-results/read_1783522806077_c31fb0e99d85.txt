   102→// --- Helpers ---

   104→function formatSalary(job: Job): string | null {
   105→  const { salaryMin, salaryMax, salaryType } = job;
   106→  if (!salaryMin && !salaryMax) return null;
   107→  const min = salaryMin ?? 0;
   108→  const max = salaryMax ?? 0;
   109→  const fmt = (n: number) =>
   110→    n >= 1000 ? `${Math.round(n / 1000)}K` : String(n);
   111→  const suffix =
   112→    salaryType === 'hourly' ? '/hr' : salaryType === 'monthly' ? '/mo' : '/yr';
   113→  const prefix = salaryType === 'hourly' ? '$' : '$';
   114→  if (min && max && min !== max) {
   115→    return `${prefix}${fmt(min)}-${fmt(max)}${suffix}`;
   116→  }
   117→  const val = max || min;
   118→  return `${prefix}${fmt(val)}${suffix}`;
   119→}
   120→
   121→function relativeDate(dateStr?: string): string {
   122→  if (!dateStr) return '';
   123→  try {
   124→    return formatDistanceToNow(new Date(dateStr), { addSuffix: true });
   125→  } catch {
   126→    return '';
   127→  }
   128→}
   129→
   130→function matchScoreColor(score: number): string {
   131→  if (score > 85)
   132→    return 'bg-emerald-100 text-emerald-700 ring-emerald-300 dark:bg-emerald-900/40 dark:text-emerald-400 dark:ring-emerald-700';
   133→  if (score > 70)
   134→    return 'bg-amber-100 text-amber-700 ring-amber-300 dark:bg-amber-900/40 dark:text-amber-400 dark:ring-amber-700';
   135→  return 'bg-gray-100 text-gray-600 ring-gray-300 dark:bg-gray-800/40 dark:text-gray-400 dark:ring-gray-600';
   136→}
