    92→    92→function relativeTime(dateStr: string): string {
    93→    93→  const now = Date.now();
    94→    94→  const then = new Date(dateStr).getTime();
    95→    95→  const diffMs = now - then;
    96→    96→  const seconds = Math.floor(diffMs / 1000);
    97→    97→  const minutes = Math.floor(seconds / 60);
    98→    98→  const hours = Math.floor(minutes / 60);
    99→    99→  const days = Math.floor(hours / 24);
   100→   100→
   101→   101→  if (seconds < 60) return 'just now';
   102→   102→  if (minutes < 60) return `${minutes} minute${minutes !== 1 ? 's' : ''} ago`;
   103→   103→  if (hours < 24) return `${hours} hour${hours !== 1 ? 's' : ''} ago`;
   104→   104→  if (days === 1) return 'yesterday';
   105→   105→  if (days < 30) return `${days} days ago`;
   106→   106→  const months = Math.floor(days / 30);
   107→   107→  return `${months} month${months !== 1 ? 's' : ''} ago`;
   108→   108→}

   110→   110→function formatDateShort(dateStr: string): string {
   111→   111→  const d = new Date(dateStr + 'T00:00:00');
   112→   112→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
