    92→function relativeTime(dateStr: string): string {
    93→  const now = Date.now();
    94→  const then = new Date(dateStr).getTime();
    95→  const diffMs = now - then;
    96→  const seconds = Math.floor(diffMs / 1000);
    97→  const minutes = Math.floor(seconds / 60);
    98→  const hours = Math.floor(minutes / 60);
    99→  const days = Math.floor(hours / 24);
   100→
   101→  if (seconds < 60) return 'just now';
   102→  if (minutes < 60) return `${minutes} minute${minutes !== 1 ? 's' : ''} ago`;
   103→  if (hours < 24) return `${hours} hour${hours !== 1 ? 's' : ''} ago`;
   104→  if (days === 1) return 'yesterday';
   105→  if (days < 30) return `${days} days ago`;
   106→  const months = Math.floor(days / 30);
   107→  return `${months} month${months !== 1 ? 's' : ''} ago`;
   108→}

   110→function formatDateShort(dateStr: string): string {
   111→  const d = new Date(dateStr + 'T00:00:00');
   112→  return d.toLocaleDateString('en-US', { month: 'short', day: 'numeric' });
   113→}
