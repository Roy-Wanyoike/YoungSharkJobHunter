   142→function formatSalary(job: JobItem): string | null {
   143→  const { salaryMin, salaryMax, salaryType } = job;
   144→  if (!salaryMin && !salaryMax) return null;
   145→  const min = salaryMin ?? 0;
   146→  const max = salaryMax ?? 0;
   147→  const fmt = (n: number) => (n >= 1000 ? `${Math.round(n / 1000)}K` : String(n));
   148→  const suffix =
   149→    salaryType === 'hourly' ? '/hr' : salaryType === 'monthly' ? '/mo' : '/yr';
   150→  if (min && max && min !== max) return `$${fmt(min)}-${fmt(max)}${suffix}`;
   151→  return `$${fmt(max || min)}${suffix}`;
   152→}
   153→
   154→function relativeDate(dateStr: string | null | undefined): string {
   155→  if (!dateStr) return '';
   156→  const diff = Date.now() - new Date(dateStr).getTime();
   157→  const days = Math.floor(diff / 86400000);
   158→  if (days === 0) return 'Today';
   159→  if (days === 1) return 'Yesterday';
   160→  if (days < 7) return `${days} days ago`;
   161→  if (days < 30) return `${Math.floor(days / 7)}w ago`;
   162→  return `${Math.floor(days / 30)}mo ago`;
   163→}
   164→
   165→function matchScoreRingColor(score: number): string {
   166→  if (score >= 80) return 'text-emerald-500';
   167→  if (score >= 60) return 'text-amber-500';
   168→  return 'text-red-500';
   169→}
   170→
   171→function matchScoreTrackColor(score: number): string {
   172→  if (score >= 80) return 'stroke-emerald-500';
   173→  if (score >= 60) return 'stroke-amber-500';
   174→  return 'stroke-red-500';
   175→}
   176→
   177→function parseSortValue(value: string): { sortBy: string; sortOrder: string } {
   178→  const [sortBy, sortOrder] = value.split('_');
   179→  return { sortBy, sortOrder };
   180→}
   181→
   182→// ─── Match Score Ring ─────────────────────────────────────────────────────────
   183→
   184→function MatchScoreRing({ score }: { score: number }) {
   185→  const radius = 18;
   186→  const circumference = 2 * Math.PI * radius;
   187→  const offset = circumference - (score / 100) * circumference;
   188→  const colorClass = matchScoreRingColor(score);
   189→  const trackClass = matchScoreTrackColor(score);
   190→
   191→  return (
   192→    <div className="relative w-12 h-12 shrink-0">
   193→      <svg className="w-12 h-12 -rotate-90" viewBox="0 0 44 44">
   194→        <circle
   195→          cx="22"
   196→          cy="22"
   197→          r={radius}
   198→          fill="none"
   199→          strokeWidth="3"
   200→          className="stroke-muted"
   201→        />
   202→        <circle
   203→          cx="22"
   204→          cy="22"
   205→          r={radius}
   206→          fill="none"
   207→          strokeWidth="3"
   208→          strokeLinecap="round"
   209→          className={trackClass}
   210→          strokeDasharray={circumference}
   211→          strokeDashoffset={offset}
   212→          style={{ transition: 'stroke-dashoffset 0.6s ease' }}
   213→        />
   214→      </svg>
   215→      <span
   216→        className={`absolute inset-0 flex items-center justify-center text-[11px] font-bold ${colorClass}`}
   217→      >
   218→        {score}
   219→      </span>
   220→    </div>
   221→  );
   222→}

   355→export default function JobsPanel() {
   356→  const [jobs, setJobs] = useState<JobItem[]>([]);
   357→  const [totalJobs, setTotalJobs] = useState(0);
   358→  const [loading, setLoading] = useState(true);
   359→  const [loadingMore, setLoadingMore] = useState(false);
   360→  const [searchQuery, setSearchQuery] = useState('');
   361→  const [debouncedSearch, setDebouncedSearch] = useState('');
   362→  const [activeFilter, setActiveFilter] = useState('all');
   363→  const [remoteOnly, setRemoteOnly] = useState(false);
   364→  const [sortValue, setSortValue] = useState('matchScore_desc');
   365→  const [page, setPage] = useState(1);
   366→  const limit = 50;
   367→
   368→  const [dialogOpen, setDialogOpen] = useState(false);
   369→  const [selectedJob, setSelectedJob] = useState<JobItem | null>(null);
   370→
   371→  const [isScraping, setIsScraping] = useState(false);
   372→  const [scrapeData, setScrapeData] = useState<ScrapeResponseData | null>(null);
   373→  const [scrapeVisible, setScrapeVisible] = useState(false);
   374→  const scrapeTimerRef = useRef<ReturnType<typeof setTimeout> | null>(null);
   375→
   376→  const { sortBy, sortOrder } = parseSortValue(sortValue);
   377→
   378→  // Debounce search
   379→  useEffect(() => {
   380→    const timer = setTimeout(() => {
   381→      setDebouncedSearch(searchQuery);
   382→    }, 300);
   383→    return () => clearTimeout(timer);
   384→  }, [searchQuery]);
   385→
   386→  // Reset page on filter/sort changes
   387→  useEffect(() => {
   388→    setPage(1);
   389→  }, [debouncedSearch, activeFilter, remoteOnly, sortValue]);
   390→
   391→  // Fetch jobs
   392→  const fetchJobs = useCallback(
   393→    async (pageNum: number, append: boolean) => {
   394→      const isFirstPage = !append;
   395→      if (isFirstPage) {
   396→        setLoading(true);
   397→      } else {
   398→        setLoadingMore(true);
   399→      }
   400→
   401→      try {
   402→        const params = new URLSearchParams({
   403→          sortBy,
   404→          sortOrder,
   405→          limit: String(limit),
   406→          page: String(pageNum),
   407→        });
   408→        if (debouncedSearch) params.set('search', debouncedSearch);
   409→        if (activeFilter !== 'all') params.set('sourceType', activeFilter);
   410→        if (remoteOnly) params.set('remote', 'true');
   411→
   412→        const res = await fetch(`/api/jobs?${params}`);
   413→        const json = await res.json();
   414→
   415→        const newJobs: JobItem[] = json.data ?? [];
   416→        const meta: JobsMeta = json.meta ?? { page: pageNum, limit, total: 0 };
   417→
   418→        if (isFirstPage) {
   419→          setJobs(newJobs);
   420→        } else {
   421→          setJobs((prev) => [...prev, ...newJobs]);
   422→        }
   423→        setTotalJobs(meta.total);
   424→      } catch {
   425→        toast.error('Failed to load jobs');
   426→      } finally {
   427→        setLoading(false);
   428→        setLoadingMore(false);
   429→      }
   430→    },
   431→    [sortBy, sortOrder, debouncedSearch, activeFilter, remoteOnly, limit]
   432→  );
   433→
   434→  // Re-fetch when page or filters change
   435→  useEffect(() => {
   436→    fetchJobs(1, false);
   437→    // eslint-disable-next-line react-hooks/exhaustive-deps
   438→  }, [debouncedSearch, activeFilter, remoteOnly, sortValue]);
   439→
   440→  const handleLoadMore = useCallback(() => {
   441→    const nextPage = page + 1;
   442→    setPage(nextPage);
   443→    fetchJobs(nextPage, true);
   444→  }, [page, fetchJobs]);
   445→
   446→  const handleToggleSave = useCallback(
   447→    async (job: JobItem) => {
   448→      const newSaved = !job.isSaved;
   449→      // Optimistic update
   450→      setJobs((prev) =>
   451→        prev.map((j) => (j.id === job.id ? { ...j, isSaved: newSaved } : j))
   452→      );
   453→      // Also update selected job if it's the same
   454→      setSelectedJob((prev) =>
   455→        prev?.id === job.id ? { ...prev, isSaved: newSaved } : prev
   456→      );
   457→
   458→      try {
   459→        await fetch('/api/jobs', {
   460→          method: 'PATCH',
   461→          headers: { 'Content-Type': 'application/json' },
   462→          body: JSON.stringify({ jobId: job.id, isSaved: newSaved }),
   463→        });
   464→        toast.success(newSaved ? 'Job saved' : 'Job unsaved');
   465→      } catch {
   466→        // Revert
   467→        setJobs((prev) =>
   468→          prev.map((j) => (j.id === job.id ? { ...j, isSaved: job.isSaved } : j))
   469→        );
   470→        setSelectedJob((prev) =>
   471→          prev?.id === job.id ? { ...prev, isSaved: job.isSaved } : prev
   472→        );
   473→        toast.error('Failed to update save status');
   474→      }
   475→    },
   476→    []
   477→  );
   478→
   479→  const handleScrape = useCallback(
   480→    async (aiOnly: boolean) => {
   481→      setIsScraping(true);
   482→      setScrapeVisible(false);
   483→      if (scrapeTimerRef.current) clearTimeout(scrapeTimerRef.current);
   484→
   485→      try {
   486→        const body = aiOnly ? { aiOnly: true } : {};
   487→        const res = await fetch('/api/scrape', {
   488→          method: 'POST',
   489→          headers: { 'Content-Type': 'application/json' },
   490→          body: JSON.stringify(body),
   491→        });
   492→        const json = await res.json();
   493→        const data: ScrapeResponseData = json.data;
   494→        setScrapeData(data);
   495→        setScrapeVisible(true);
   496→
   497→        // Auto-dismiss after 10s
   498→        scrapeTimerRef.current = setTimeout(() => {
   499→          setScrapeVisible(false);
   500→        }, 10000);
   501→
   502→        toast.success(
   503→          `Scraped ${data.sourcesScraped} sources — ${data.totalJobsNew} new jobs found`
   504→        );
   505→
   506→        // Refresh jobs
   507→        setPage(1);
   508→        fetchJobs(1, false);
   509→      } catch {
   510→        toast.error('Scraping failed. Please try again.');
   511→      } finally {
   512→        setIsScraping(false);
   513→      }
   514→    },
   515→    [fetchJobs]
   516→  );
   517→
   518→  const dismissScrapeBanner = useCallback(() => {
   519→    setScrapeVisible(false);
   520→    if (scrapeTimerRef.current) clearTimeout(scrapeTimerRef.current);
   521→  }, []);
   522→
   523→  const hasMore = jobs.length < totalJobs;
   524→  const displayedSourceLabel =
   525→    SOURCE_TYPE_LABEL[activeFilter] ?? activeFilter.replace(/_/g, ' ');
   526→
   527→  return (
   528→    <div className="space-y-4">
   529→      {/* ── Scrape Buttons ── */}
   530→      <div className="flex flex-wrap gap-2">
   531→        <Button
   532→          onClick={() => handleScrape(false)}
   533→          disabled={isScraping}
   534→          className="text-xs"
   535→        >
   536→          {isScraping ? (
   537→            <Loader2 className="w-3.5 h-3.5 animate-spin mr-1.5" />
   538→          ) : (
   539→            <RefreshCw className="w-3.5 h-3.5 mr-1.5" />
   540→          )}
   541→          Scrape All Sources
   542→        </Button>
   543→        <Button
   544→          variant="outline"
   545→          onClick={() => handleScrape(true)}
   546→          disabled={isScraping}
   547→          className="text-xs"
   548→        >
   549→          {isScraping ? (
   550→            <Loader2 className="w-3.5 h-3.5 animate-spin mr-1.5" />
   551→          ) : (
   552→            <RefreshCw className="w-3.5 h-3.5 mr-1.5" />
   553→          )}
   554→          Scrape AI Training Only
   555→        </Button>
   556→      </div>
   557→
   558→      {/* ── Scrape Results Banner ── */}
   559→      {scrapeVisible && scrapeData && (
   560→        <Card className="border-emerald-200 dark:border-emerald-800 relative">
   561→          <button
   562→            onClick={dismissScrapeBanner}
   563→            className="absolute top-3 right-3 p-1 rounded-md hover:bg-muted transition-colors"
   564→            aria-label="Dismiss results"
   565→          >
   566→            <X className="w-4 h-4" />
   567→          </button>
   568→          <CardContent className="p-3">
   569→            <div className="flex items-center gap-2 mb-2">
   570→              <CheckCircle2 className="w-4 h-4 text-emerald-500" />
   571→              <p className="text-xs font-medium text-emerald-600 dark:text-emerald-400">
   572→                Scraping Complete — {scrapeData.totalJobsFound} found,{' '}
   573→                {scrapeData.totalJobsNew} new
   574→              </p>
   575→            </div>
   576→            <div className="overflow-x-auto max-h-48 overflow-y-auto">
   577→              <table className="w-full text-sm">
   578→                <thead>
   579→                  <tr className="border-b border-border">
   580→                    <th className="text-left py-1 pr-4 font-medium text-muted-foreground text-xs">
   581→                      Source
   582→                    </th>
   583→                    <th className="text-right py-1 px-3 font-medium text-muted-foreground text-xs">
