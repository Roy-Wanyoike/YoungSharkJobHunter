   142→   142→function formatSalary(job: JobItem): string | null {
   143→   143→  const { salaryMin, salaryMax, salaryType } = job;
   144→   144→  if (!salaryMin && !salaryMax) return null;
   145→   145→  const min = salaryMin ?? 0;
   146→   146→  const max = salaryMax ?? 0;
   147→   147→  const fmt = (n: number) => (n >= 1000 ? `${Math.round(n / 1000)}K` : String(n));
   148→   148→  const suffix =
   149→   149→    salaryType === 'hourly' ? '/hr' : salaryType === 'monthly' ? '/mo' : '/yr';
   150→   150→  if (min && max && min !== max) return `$${fmt(min)}-${fmt(max)}${suffix}`;
   151→   151→  return `$${fmt(max || min)}${suffix}`;
   152→   152→}
   153→   153→
   154→   154→function relativeDate(dateStr: string | null | undefined): string {
   155→   155→  if (!dateStr) return '';
   156→   156→  const diff = Date.now() - new Date(dateStr).getTime();
   157→   157→  const days = Math.floor(diff / 86400000);
   158→   158→  if (days === 0) return 'Today';
   159→   159→  if (days === 1) return 'Yesterday';
   160→   160→  if (days < 7) return `${days} days ago`;
   161→   161→  if (days < 30) return `${Math.floor(days / 7)}w ago`;
   162→   162→  return `${Math.floor(days / 30)}mo ago`;
   163→   163→}

   165→   165→function matchScoreRingColor(score: number): string {
   166→   166→  if (score >= 80) return 'text-emerald-500';
   167→   167→  if (score >= 60) return 'text-amber-500';
   168→   168→  return 'text-red-500';
   169→   169→}
   170→   170→
   171→   171→function matchScoreTrackColor(score: number): string {
   172→   172→  if (score >= 80) return 'stroke-emerald-500';
   173→   173→  if (score >= 60) return 'stroke-amber-500';
   174→   174→  return 'stroke-red-500';
   175→   175→}

   177→   177→function parseSortValue(value: string): { sortBy: string; sortOrder: string } {
   178→   178→  const [sortBy, sortOrder] = value.split('_');
   179→   179→  return { sortBy, sortOrder };
   180→   180→}

   184→   184→function MatchScoreRing({ score }: { score: number }) {
   185→   185→  const radius = 18;
   186→   186→  const circumference = 2 * Math.PI * radius;
   187→   187→  const offset = circumference - (score / 100) * circumference;
   188→   188→  const colorClass = matchScoreRingColor(score);
   189→   189→  const trackClass = matchScoreTrackColor(score);
   190→   190→
   191→   191→  return (
   192→   192→    <div className="relative w-12 h-12 shrink-0">
   193→   193→      <svg className="w-12 h-12 -rotate-90" viewBox="0 0 44 44">
   194→   194→        <circle
   195→   195→          cx="22"
   196→   196→          cy="22"
   197→   197→          r={radius}
   198→   198→          fill="none"
   199→   199→          strokeWidth="3"
   200→   200→          className="stroke-muted"
   201→   201→        />
   202→   202→        <circle
   203→   203→          cx="22"
   204→   204→          cy="22"
   205→   205→          r={radius}
   206→   206→          fill="none"
   207→   207→          strokeWidth="3"
   208→   208→          strokeLinecap="round"
   209→   209→          className={trackClass}
   210→   210→          strokeDasharray={circumference}
   211→   211→          strokeDashoffset={offset}
   212→   212→          style={{ transition: 'stroke-dashoffset 0.6s ease' }}
   213→   213→        />
   214→   214→      </svg>
   215→   215→      <span
   216→   216→        className={`absolute inset-0 flex items-center justify-center text-[11px] font-bold ${colorClass}`}
   217→   217→      >
   218→   218→        {score}
   219→   219→      </span>
   220→   220→    </div>
   221→   221→  );
   222→   222→}

   362→   362→  const [activeFilter, setActiveFilter] = useState('all');

   364→   364→  const [sortValue, setSortValue] = useState('matchScore_desc');

   375→   375→

   377→   377→
   378→   378→  // Debounce search
   379→   379→  useEffect(() => {
   380→   380→    const timer = setTimeout(() => {
   381→   381→      setDebouncedSearch(searchQuery);
   382→   382→    }, 300);
   383→   383→    return () => clearTimeout(timer);
   384→   384→  }, [searchQuery]);
   385→   385→
   386→   386→  // Reset page on filter/sort changes
   387→   387→  useEffect(() => {
   388→   388→    setPage(1);
   389→   389→  }, [debouncedSearch, activeFilter, remoteOnly, sortValue]);
   390→   390→
   391→   391→  // Fetch jobs
   392→   392→  const fetchJobs = useCallback(
   393→   393→    async (pageNum: number, append: boolean) => {
   394→   394→      const isFirstPage = !append;
   395→   395→      if (isFirstPage) {
   396→   396→        setLoading(true);

   398→   398→        setLoadingMore(true);

   400→   400→

   403→   403→          sortBy,
   404→   404→          sortOrder,
   405→   405→          limit: String(limit),

   408→   408→        if (debouncedSearch) params.set('search', debouncedSearch);
   409→   409→        if (activeFilter !== 'all') params.set('sourceType', activeFilter);
   410→   410→        if (remoteOnly) params.set('remote', 'true');
   411→   411→

   413→   413→        const json = await res.json();
   414→   414→

   416→   416→        const meta: JobsMeta = json.meta ?? { page: pageNum, limit, total: 0 };
   417→   417→
   418→   418→        if (isFirstPage) {
   419→   419→          setJobs(newJobs);
   420→   420→        } else {
   421→   421→          setJobs((prev) => [...prev, ...newJobs]);
   422→   422→        }
   423→   423→        setTotalJobs(meta.total);
   424→   424→      } catch {
   425→   425→        toast.error('Failed to load jobs');
   426→   426→      } finally {
   427→   427→        setLoading(false);
   428→   428→        setLoadingMore(false);
   429→   429→      }
   430→   430→    },
   431→   431→    [sortBy, sortOrder, debouncedSearch, activeFilter, remoteOnly, limit]
   432→   432→  );
   433→   433→
   434→   434→  // Re-fetch when page or filters change
   435→   435→  useEffect(() => {
   436→   436→    fetchJobs(1, false);
   437→   437→    // eslint-disable-next-line react-hooks/exhaustive-deps
   438→   438→  }, [debouncedSearch, activeFilter, remoteOnly, sortValue]);
   439→   439→
   440→   440→  const handleLoadMore = useCallback(() => {
   441→   441→    const nextPage = page + 1;
   442→   442→    setPage(nextPage);
   443→   443→    fetchJobs(nextPage, true);
   444→   444→  }, [page, fetchJobs]);
   445→   445→
   446→   446→  const handleToggleSave = useCallback(
   447→   447→    async (job: JobItem) => {
   448→   448→      const newSaved = !job.isSaved;
   449→   449→      // Optimistic update
   450→   450→      setJobs((prev) =>
   451→   451→        prev.map((j) => (j.id === job.id ? { ...j, isSaved: newSaved } : j))
   452→   452→      );
   453→   453→      // Also update selected job if it's the same
   454→   454→      setSelectedJob((prev) =>
   455→   455→        prev?.id === job.id ? { ...prev, isSaved: newSaved } : prev
   456→   456→      );
   457→   457→
   458→   458→      try {
   459→   459→        await fetch('/api/jobs', {
   460→   460→          method: 'PATCH',
   461→   461→          headers: { 'Content-Type': 'application/json' },
   462→   462→          body: JSON.stringify({ jobId: job.id, isSaved: newSaved }),
   463→   463→        });
   464→   464→        toast.success(newSaved ? 'Job saved' : 'Job unsaved');
   465→   465→      } catch {
   466→   466→        // Revert
   467→   467→        setJobs((prev) =>
   468→   468→          prev.map((j) => (j.id === job.id ? { ...j, isSaved: job.isSaved } : j))
   469→   469→        );
   470→   470→        setSelectedJob((prev) =>
   471→   471→          prev?.id === job.id ? { ...prev, isSaved: job.isSaved } : prev
   472→   472→        );
   473→   473→        toast.error('Failed to update save status');
   474→   474→      }
   475→   475→    },
   476→   476→    []
   477→   477→  );
   478→   478→
   479→   479→  const handleScrape = useCallback(
   480→   480→    async (aiOnly: boolean) => {
   481→   481→      setIsScraping(true);
   482→   482→      setScrapeVisible(false);
