    76→    76→
    77→    77→function ATSGauge({ score, label, color }: { score: number; label: string; color: string }) {
    78→    78→  const radius = 54;
    79→    79→  const circumference = 2 * Math.PI * radius;
    80→    80→  const offset = circumference - (score / 100) * circumference;
    81→    81→
    82→    82→  return (
    83→    83→    <div className="flex flex-col items-center gap-1.5">
    84→    84→      <div className="relative h-32 w-32">
    85→    85→        <svg className="h-full w-full -rotate-90" viewBox="0 0 120 120">
    86→    86→          <circle
    87→    87→            cx="60"
    88→    88→            cy="60"
    89→    89→            r={radius}
    90→    90→            fill="none"
    91→    91→            stroke="currentColor"
    92→    92→            strokeWidth="8"
    93→    93→            className="text-muted/30"
    94→    94→          />
    95→    95→          <circle
    96→    96→            cx="60"
    97→    97→            cy="60"
    98→    98→            r={radius}
    99→    99→            fill="none"
   100→   100→            stroke={color}
   101→   101→            strokeWidth="8"
   102→   102→            strokeLinecap="round"
   103→   103→            strokeDasharray={circumference}
   104→   104→            strokeDashoffset={offset}
   105→   105→            className="transition-all duration-1000 ease-out"
   106→   106→          />
   107→   107→        </svg>
   108→   108→        <div className="absolute inset-0 flex flex-col items-center justify-center">
   109→   109→          <span className="text-3xl font-bold" style={{ color }}>{score}%</span>
   110→   110→        </div>
   111→   111→      </div>
   112→   112→      <span className="text-xs font-medium text-muted-foreground">{label}</span>
   113→   113→    </div>
   114→   114→  );
   115→   115→}
