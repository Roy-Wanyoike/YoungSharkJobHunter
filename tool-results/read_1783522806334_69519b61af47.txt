   148→// Deterministic hue per company name for avatar colour
   149→function companyHue(name: string): string {
   150→  let hash = 0;
   151→  for (let i = 0; i < name.length; i++) {
   152→    hash = name.charCodeAt(i) + ((hash << 5) - hash);
   153→  }
   154→  const hue = Math.abs(hash) % 360;
   155→  return `hsl(${hue}, 55%, 50%)`;
   156→}

   162→function CircularAtsScore({ score }: { score: number }) {
   163→  const radius = 18;
   164→  const circumference = 2 * Math.PI * radius;
   165→  const offset = circumference - (score / 100) * circumference;
   166→
   167→  const colour =
   168→    score >= 80
   169→      ? 'text-emerald-500'
   170→      : score >= 60
   171→        ? 'text-amber-500'
   172→        : 'text-red-500';
   173→
   174→  return (
   175→    <div className="relative inline-flex size-10 items-center justify-center">
   176→      <svg className="size-10 -rotate-90" viewBox="0 0 44 44">
   177→        <circle
   178→          cx="22"
   179→          cy="22"
   180→          r={radius}
   181→          fill="none"
   182→          stroke="currentColor"
   183→          strokeWidth="3"
   184→          className="text-muted/30"
   185→        />
   186→        <circle
   187→          cx="22"
   188→          cy="22"
   189→          r={radius}
   190→          fill="none"
   191→          stroke="currentColor"
   192→          strokeWidth="3"
   193→          strokeLinecap="round"
   194→          strokeDasharray={circumference}
   195→          strokeDashoffset={offset}
   196→          className={colour}
   197→        />
   198→      </svg>
   199→      <span className={`absolute text-[10px] font-bold ${colour}`}>
   200→        {score}
   201→      </span>
   202→    </div>
   203→  );
   204→}

   550→function StatsSummary({ applications }: { applications: Application[] }) {
   551→  const total = applications.length;
   552→  const responded = applications.filter(
   553→    (a) =>
   554→      a.status === 'reviewing' ||
   555→      a.status === 'interview' ||
   556→      a.status === 'assessment' ||
   557→      a.status === 'offer' ||
   558→      a.status === 'rejected'
   559→  ).length;
   560→  const interviewed = applications.filter(
   561→    (a) =>
   562→      a.status === 'interview' ||
   563→      a.status === 'assessment' ||
   564→      a.status === 'offer'
   565→  ).length;
   566→  const offered = applications.filter(
   567→    (a) => a.status === 'offer'
   568→  ).length;
   569→
   570→  const stats = [
   571→    {
   572→      label: 'Total Applications',
   573→      value: total,
   574→      icon: Briefcase,
   575→      color: 'text-foreground',
   576→    },
   577→    {
   578→      label: 'Response Rate',
   579→      value: total > 0 ? `${Math.round((responded / total) * 100)}%` : '0%',
   580→      icon: TrendingUp,
   581→      color: 'text-teal-600 dark:text-teal-400',
   582→    },
   583→    {
   584→      label: 'Interview Rate',
   585→      value:
   586→        total > 0 ? `${Math.round((interviewed / total) * 100)}%` : '0%',
   587→      icon: Video,
   588→      color: 'text-emerald-600 dark:text-emerald-400',
   589→    },
   590→    {
   591→      label: 'Offer Rate',
   592→      value: total > 0 ? `${Math.round((offered / total) * 100)}%` : '0%',
   593→      icon: Award,
   594→      color: 'text-green-600 dark:text-green-400',
   595→    },
   596→  ];
   597→
   598→  return (
   599→    <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   600→      {stats.map((s) => (
   601→        <Card key={s.label} className="py-0 gap-0">
   602→          <CardContent className="p-4 flex items-center gap-3">
   603→            <div
   604→              className={`rounded-lg bg-muted/60 p-2 ${s.color}`}
   605→            >
   606→              <s.icon className="size-4" />
   607→            </div>
   608→            <div>
   609→              <p className="text-lg font-bold leading-tight">{s.value}</p>
   610→              <p className="text-[11px] text-muted-foreground leading-tight">
   611→                {s.label}
   612→              </p>
   613→            </div>
   614→          </CardContent>
   615→        </Card>
   616→      ))}
   617→    </div>
   618→  );
   619→}

   630→  const [applications, setApplications] = useState<Application[]>([]);
   631→  const [statusCounts, setStatusCounts] = useState<StatusCounts>({
   632→    all: 0,
   633→    submitted: 0,
   634→    reviewing: 0,
   635→    interview: 0,
   636→    assessment: 0,
   637→    offer: 0,
   638→    rejected: 0,
   639→    ghosted: 0,
   640→  });
   641→  const [loading, setLoading] = useState(true);
   642→
   643→  const fetchApplications = useCallback(async () => {
   644→    setLoading(true);
   645→    try {
   646→      const query =
   647→        selectedFilter === 'all' ? '' : `?status=${selectedFilter}`;
   648→      const res = await fetch(`/api/applications${query}`);
   649→      if (!res.ok) throw new Error('Failed to fetch applications');
   650→      const data = await res.json();
   651→
   652→      const apps: Application[] = data.applications ?? data ?? [];
   653→      const counts: StatusCounts = data.counts ?? {
   654→        all: 0,
   655→        submitted: 0,
   656→        reviewing: 0,
   657→        interview: 0,
   658→        assessment: 0,
   659→        offer: 0,
   660→        rejected: 0,
   661→        ghosted: 0,
   662→      };
   663→
   664→      // If counts not returned, compute them ourselves
   665→      if (!data.counts) {
   666→        const computed: Record<string, number> = {
   667→          all: apps.length,
   668→          submitted: 0,
   669→          reviewing: 0,
   670→          interview: 0,
   671→          assessment: 0,
   672→          offer: 0,
   673→          rejected: 0,
   674→          ghosted: 0,
   675→        };
   676→        for (const a of apps) {
   677→          if (computed[a.status] !== undefined) {
   678→            computed[a.status]++;
   679→          }
   680→        }
   681→        counts.all = computed.all;
   682→        counts.submitted = computed.submitted;
   683→        counts.reviewing = computed.reviewing;
   684→        counts.interview = computed.interview;
   685→        counts.assessment = computed.assessment;
   686→        counts.offer = computed.offer;
   687→        counts.rejected = computed.rejected;
   688→        counts.ghosted = computed.ghosted;
   689→      }
   690→
   691→      setApplications(apps);
   692→      setStatusCounts(counts);
   693→    } catch {
   694→      // On error, show empty state — silent fallback
   695→      setApplications([]);
   696→    } finally {
   697→      setLoading(false);
   698→    }
   699→  }, [selectedFilter]);
   700→
   701→  useEffect(() => {
   702→    fetchApplications();
   703→  }, [fetchApplications]);
   704→
   705→  const filteredApps = useMemo(() => {
   706→    if (selectedFilter === 'all') return applications;
   707→    return applications.filter((a) => a.status === selectedFilter);
   708→  }, [applications, selectedFilter]);
   709→
   710→  return (
   711→    <div className="flex flex-col gap-4 h-full">
   712→      {/* Header + View Toggle */}
   713→      <div className="flex items-center justify-between flex-wrap gap-3">
   714→        <h2 className="text-lg font-semibold flex items-center gap-2">
   715→          <Briefcase className="size-5" />
   716→          Applications
   717→        </h2>
   718→        <div className="flex items-center gap-1 bg-muted rounded-lg p-1">
   719→          <Button
   720→            variant={viewMode === 'card' ? 'default' : 'ghost'}
   721→            size="sm"
   722→            className="h-7 px-2.5 text-xs"
   723→            onClick={() => setViewMode('card')}
   724→          >
   725→            <LayoutGrid className="size-3.5 mr-1" />
   726→            Card View
   727→          </Button>
   728→          <Button
   729→            variant={viewMode === 'pipeline' ? 'default' : 'ghost'}
   730→            size="sm"
   731→            className="h-7 px-2.5 text-xs"
   732→            onClick={() => setViewMode('pipeline')}
   733→          >
   734→            <Columns className="size-3.5 mr-1" />
   735→            Pipeline View
   736→          </Button>
   737→        </div>
   738→      </div>
   739→
   740→      {/* Status Filter Bar */}
   741→      <ScrollArea className="w-full">
   742→        <div className="flex gap-2 pb-2 min-w-max">
   743→          {ALL_STATUSES.map((s) => {
   744→            const isActive = selectedFilter === s;
   745→            const label =
   746→              s === 'all'
   747→                ? 'All'
   748→                : STATUS_CONFIG[s as ApplicationStatus]?.label ?? s;
   749→            const count =
   750→              s === 'all'
   751→                ? statusCounts.all
   752→                : statusCounts[s as ApplicationStatus] ?? 0;
   753→
   754→            return (
   755→              <Button
   756→                key={s}
   757→                variant={isActive ? 'default' : 'outline'}
   758→                size="sm"
   759→                className="h-8 text-xs gap-1.5 rounded-full"
   760→                onClick={() => setSelectedFilter(s)}
   761→              >
   762→                {label}
   763→                <Badge
   764→                  variant="secondary"
   765→                  className={`ml-0.5 h-5 min-w-5 px-1.5 text-[10px] ${
   766→                    isActive
   767→                      ? 'bg-primary-foreground/20 text-primary-foreground'
   768→                      : ''
   769→                  }`}
   770→                >
   771→                  {count}
   772→                </Badge>
   773→              </Button>
   774→            );
   775→          })}
   776→        </div>
   777→        <ScrollBar orientation="horizontal" />
   778→      </ScrollArea>
   779→
   780→      {/* Content */}
   781→      <div className="flex-1 min-h-0">
   782→        {loading ? (
   783→          <div className="space-y-3">
   784→            {Array.from({ length: 4 }).map((_, i) => (
   785→              <SkeletonCard key={i} />
   786→            ))}
   787→          </div>
   788→        ) : filteredApps.length === 0 ? (
   789→          <div className="flex flex-col items-center justify-center py-16 text-center">
   790→            <div className="rounded-full bg-muted p-4 mb-4">
   791→              <Briefcase className="size-8 text-muted-foreground" />
   792→            </div>
   793→            <h3 className="font-medium text-sm">No applications found</h3>
   794→            <p className="text-xs text-muted-foreground mt-1">
   795→              {selectedFilter === 'all'
   796→                ? 'Start tracking your job applications.'
   797→                : `No applications with "${STATUS_CONFIG[selectedFilter as ApplicationStatus]?.label}" status.`}
   798→            </p>
   799→          </div>
   800→        ) : viewMode === 'card' ? (
   801→          <ScrollArea className="h-full">
   802→            <div className="space-y-3 pr-3">
   803→              <AnimatePresence mode="popLayout">
   804→                {filteredApps.map((app) => (
   805→                  <ApplicationCard key={app.id} app={app} />
   806→                ))}
   807→              </AnimatePresence>
   808→            </div>
   809→          </ScrollArea>
   810→        ) : (
   811→          <PipelineView applications={applications} />
   812→        )}
   813→      </div>
   814→
   815→      {/* Stats Summary */}
   816→      {!loading && applications.length > 0 && (
   817→        <Separator />
   818→      )}
   819→      {!loading && applications.length > 0 && (
   820→        <StatsSummary applications={applications} />
   821→      )}
   822→    </div>
   823→  );
   824→}
